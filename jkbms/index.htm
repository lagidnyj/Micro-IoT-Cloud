function formatStorageTimestamp(rawValue) { if (!rawValue) return '—'; const normalizedValue = String(rawValue).trim(); let date; if (/^\d+$/.test(normalizedValue)) { const timestamp = Number(normalizedValue); date = new Date(timestamp * 1000); } else { date = new Date(normalizedValue); } if (Number.isNaN(date.getTime())) { return normalizedValue; } const localTime = new Intl.DateTimeFormat('uk-UA', { timeZone: Intl.DateTimeFormat().resolvedOptions().timeZone || 'UTC', year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false, }).format(date); const utcTime = new Date(date).toISOString() .replace('T', ' ') .replace(/\.\d{3}Z$/, ' UTC'); return `${localTime} (UTC: ${utcTime})`; }
